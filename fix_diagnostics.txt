  79:     def add(self, raw_text, conf, observation_id=None):

  81:         if observation_id is not None:
  82:             if observation_id in self._observation_ids:
  83:                 return False
  84:             self._observation_ids.add(observation_id)
  85:         fixed, valid = fix_indian(raw_text)
  86: 
  87:         if len(fixed) < 6:
  88:             return False
  89: 
  90:         # Reject structurally invalid standard Indian plates.
  91:         # Standard format:
  92:         #   STATE + 1-2 district digits + 1-3 series letters + 4 digits.
  93:         # Bharat (BH) plates keep their existing validation path.
  94:         if valid and fixed[2:4] != "BH":
  95:             if len(fixed) not in (9, 10):
  96:                 return False
  97: 
  98:             state = fixed[:2]
  99:             body = fixed[2:-4]
 100:             suffix = fixed[-4:]

 102:             if state not in STATE_CODES or not suffix.isdigit():
 103:                 return False
 104: 
 105:             district_len = 0
 106:             while (
 107:                 district_len < min(2, len(body))
 108:                 and body[district_len].isdigit()
 109:             ):
 110:                 district_len += 1
 111: 
 112:             series = body[district_len:]

 114:             if (

 117:                 or not series.isalpha()
 118:             ):
 119:                 return False
 120: 
 121:         self.reads.append((fixed, float(conf), valid))
 122: 
 123:         if len(self.reads) > self.max_reads:
 124:             self.reads.pop(0)
 125:         return True
 126: 
 127:     def consensus(self):
 128:         """Return the strongest exact normalized registration, never a synthetic
 129:         plate assembled by mixing characters from different registrations."""
 130:         if not self.reads:
 131:             return None
 132: 
 133:         valid_reads = [r for r in self.reads if r[2]]
 134:         pool = valid_reads if valid_reads else self.reads
 135:         grouped = {}
 136:         for text, conf, valid in pool:
 137:             item = grouped.setdefault(text, {"confs": [], "valid": valid})
 138:             item["confs"].append(float(conf))
 139: 
 140:         # Weight support by both repeat count and OCR confidence. Stable sort
 141:         # makes ties deterministic; ties are still not enough to confirm.
 142:         text, item = max(
 143:             grouped.items(),
 144:             key=lambda pair: (
 145:                 sum(pair[1]["confs"]),
 146:                 len(pair[1]["confs"]),
 147:                 pair[0],
 148:             ),
 149:         )
 150:         confs = item["confs"]
 151:         return Consensus(
 152:             text=text,
 153:             conf=sum(confs) / len(confs),
 154:             n_reads=len(confs),
 155:             valid=bool(item["valid"]),
 156:         )
 157: 
 158:     def is_final(self, confirm_reads, confirm_conf):
 159:         # Confirm only a single normalized full registration supported by
 160:         # enough observations. Do not merge unrelated plates character-wise.
 161:         by_text = {}
 162:         for text, conf, valid in self.reads:
 163:             if valid:
 164:                 by_text.setdefault(text, []).append(float(conf))
 165: 
 166:         eligible = [
 167:             (plate, confs)
 168:             for plate, confs in by_text.items()
 169:             if len(confs) >= confirm_reads
 170:             and sum(confs) / len(confs) >= confirm_conf
 171:         ]
 172: 
 173:         # Ambiguous competing registrations must never be auto-confirmed.
 174:         if len(eligible) != 1:
 175:             return False
 176: 
 177:         return True

 180: def crop_quality(img_bgr, min_brightness=8.0, max_brightness=247.0,
 181:                 min_contrast=5.0, min_laplacian_var=2.0):
 182:     """Conservative quality gate for vehicle crops passed to the plate detector.
 183: 
 184:     Returns (accepted, metrics). Thresholds are configurable; this is a
 185:     rejection gate, not a claim that the crop contains a readable plate.
 186:     """
 187:     import cv2
 188:     import numpy as np
 189: 
 190:     if img_bgr is None or not getattr(img_bgr, "size", 0):
 191:         return False, {"reason": "empty"}
 192:     if len(img_bgr.shape) < 2 or img_bgr.shape[0] < 16 or img_bgr.shape[1] < 16:
 193:         return False, {"reason": "too_small"}
 194: 
 195:     gray = cv2.cvtColor(img_bgr, cv2.COLOR_BGR2GRAY) if len(img_bgr.shape) == 3 else img_bgr
 196:     brightness = float(np.mean(gray))
 197:     contrast = float(np.std(gray))
 198:     sharpness = float(cv2.Laplacian(gray, cv2.CV_64F).var())
 199:     metrics = {
 200:         "brightness": round(brightness, 2),
 201:         "contrast": round(contrast, 2),
 202:         "laplacian_var": round(sharpness, 2),
 203:     }
 204:     if brightness < min_brightness or brightness > max_brightness:
 205:         metrics["reason"] = "brightness"
 206:         return False, metrics
 207:     if contrast < min_contrast:
 208:         metrics["reason"] = "contrast"
 209:         return False, metrics
 210:     if sharpness < min_laplacian_var:
 211:         metrics["reason"] = "blur"
 212:         return False, metrics
 213:     return True, metrics
